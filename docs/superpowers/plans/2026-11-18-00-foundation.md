# Foundation (Prompt 00) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Establish a running, testable skeleton with auth plumbing, shared shell layout, mockable Docker layer, and Docker Compose orchestration.

**Architecture:** Decoupled FastAPI backend with Async SQLAlchemy 2 + Alembic, Pydantic settings loading `config/app.yaml`, and a `DockerGateway` interface with an in-memory `FakeGateway`. Standalone Angular 19 frontend with signal-based stores, auto-reconnecting `SseService`, Angular Material CSS-variable theme, and dynamic navigation shell. Docker Compose tying services together with `docker-socket-proxy`.

**Tech Stack:** Python 3.12+, FastAPI, SQLAlchemy 2 (async), asyncpg, Alembic, pytest, Angular 19, TypeScript, Angular Material, RxJS, Docker Compose, PostgreSQL.

**Spec:** [docs/superpowers/specs/2026-11-18-00-foundation-design.md](file:///c:/Santosh/Development/dockpilot/docs/superpowers/specs/2026-11-18-00-foundation-design.md)

## Global Constraints

- No hardcoded domain terms anywhere: thresholds, intervals, container names, labels, roles, colours and route titles all come from `config/*.yaml`, the database or a typed constants file.
- Every backend feature has a service layer, a router, schemas and tests. The Docker client sits behind an abstract interface so tests use a fake.
- Every Angular page is a lazy-loaded standalone route with a typed API service, loading, empty and error states, and a unit test.
- Every control action writes an audit record before it runs, then updates it with the result.
- Role checks (viewer, operator, admin) are enforced in the backend first and mirrored in the UI.
- Never expose the backend publicly. Default bind is `127.0.0.1`.
- Finish each prompt with: tests passing, a short README section for the page, and a stop. Wait for the next prompt.

```python
    port: int = 8000
    database_url: str = "postgresql+asyncpg://dockpilot:dockpilot@127.0.0.1:5432/dockpilot"

    jwt_secret: str = "dev-secret-key-change-in-production-min-32-chars-long"
    jwt_access_expire_minutes: int = 15
    jwt_refresh_expire_days: int = 7

@lru_cache
def get_settings() -> Settings:
    config_path = Path(__file__).resolve().parents[3] / "config" / "app.yaml"
    if not config_path.exists():
        config_path = Path("config/app.yaml")
    if config_path.exists():
        with open(config_path, "r", encoding="utf-8") as f:
            data = yaml.safe_load(f) or {}
            return Settings(**data)
    return Settings()
```

1. Write `backend/app/docker/interface.py`:
```python
from abc import ABC, abstractmethod
from typing import AsyncGenerator, Dict, List, Any

class DockerGateway(ABC):
    @abstractmethod
    async def ping(self) -> bool:
        pass

    @abstractmethod
    async def list_containers(self, all: bool = True) -> List[Dict[str, Any]]:
        pass

    @abstractmethod
    async def inspect_container(self, container_id: str) -> Dict[str, Any]:
        pass

    @abstractmethod
    async def execute_action(self, container_id: str, action: str) -> bool:
        pass

    @abstractmethod
    async def stream_events(self) -> AsyncGenerator[Dict[str, Any], None]:
        pass
```

2. Write `backend/app/docker/fake.py`:
```python
from typing import AsyncGenerator, Dict, List, Any
from app.docker.interface import DockerGateway

class FakeGateway(DockerGateway):
    def __init__(self):
        self.containers = {
            "c1": {
                "id": "c1",
                "name": "dockpilot-redis",
                "image": "redis:7-alpine",
                "state": "running",
                "status": "Up 2 hours",
                "created": "2026-11-18T20:00:00Z",
                "ports": {"6379/tcp": [{"HostIp": "0.0.0.0", "HostPort": "6379"}]},
                "labels": {"com.docker.compose.project": "dockpilot"},
                "env": ["REDIS_PASSWORD=secret123", "PORT=6379"]
            }
        }

    async def ping(self) -> bool:
        return True

    async def list_containers(self, all: bool = True) -> List[Dict[str, Any]]:
        return list(self.containers.values())

    async def inspect_container(self, container_id: str) -> Dict[str, Any]:
        if container_id in self.containers:
            return self.containers[container_id]
        raise ValueError(f"Container {container_id} not found")

    async def execute_action(self, container_id: str, action: str) -> bool:
        if container_id in self.containers:
            if action in ["start", "restart", "unpause"]:
                self.containers[container_id]["state"] = "running"
            elif action in ["stop", "kill"]:
                self.containers[container_id]["state"] = "exited"
            elif action == "pause":
                self.containers[container_id]["state"] = "paused"
            return True
        return False

    async def stream_events(self) -> AsyncGenerator[Dict[str, Any], None]:
        yield {"status": "start", "id": "c1", "from": "redis:7-alpine"}
```

```typescript
import { Injectable, signal } from '@angular/core';
import { Observable } from 'rxjs';

export type SseConnectionState = 'connected' | 'reconnecting' | 'disconnected';

@Injectable({ providedIn: 'root' })
export class SseService {
  readonly connectionState = signal<SseConnectionState>('disconnected');
  private eventSource: EventSource | null = null;
  private retryCount = 0;

  computeBackoffDelay(retry: number): number {
    const base = 1000;
    const factor = 1.5;
    const max = 30000;
    return Math.min(base * Math.pow(factor, retry), max);
  }

  connect<T>(url: string): Observable<T> {
    return new Observable<T>(observer => {
      this.eventSource = new EventSource(url);
      this.connectionState.set('connected');

      this.eventSource.onmessage = (event) => {
        try {
          const parsed = JSON.parse(event.data);
          observer.next(parsed);
        } catch {
          observer.next(event.data as unknown as T);
        }
      };

      this.eventSource.onerror = () => {
        this.connectionState.set('reconnecting');
        this.eventSource?.close();
        this.retryCount++;
      };

      return () => {
        this.eventSource?.close();
        this.connectionState.set('disconnected');
        this.retryCount = 0;
      };
    });
  }
}
```

```yaml
version: "3.8"

    image: tecnativa/docker-socket-proxy:latest
```
